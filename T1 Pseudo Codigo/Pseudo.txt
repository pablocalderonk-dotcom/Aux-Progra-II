EJERCICIO # 2
Inicio
    Clase Laptop
        Atributos: fabricante, cpu, memoria_ram, disco
        Método Constructor(fab, c, mem, d)
            fabricante <- fab
            cpu <- c
            memoria_ram <- mem
            disco <- d
        Fin Método

        Método comprobar_ram(valor)
            Retornar (memoria_ram == valor)
        Fin Método

        Método visualizar()
            Imprimir fabricante, cpu, memoria_ram, disco
        Fin Método
    Fin Clase

    laptop1 <- Nuevo Laptop("Lenovo", "Intel Core i7", 16, 512)
    laptop2 <- Nuevo Laptop(fabricante="Acer", cpu="AMD Ryzen 5", memoria_ram=8, disco=1024)

    R <- 16
    Si laptop1.comprobar_ram(R) Entonces
        Imprimir laptop1
    Fin Si

    Si laptop1.disco > laptop2.disco Entonces
        laptop1.visualizar()
    Sino
        laptop2.visualizar()
    Fin Si
Fin
